cmake_minimum_required(VERSION 3.22)
project(NFCapture VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS")
add_subdirectory(JUCE)
juce_add_binary_data(NFAssets SOURCES Assets/Logo.png)
set(formats VST3)
if(APPLE)
 list(APPEND formats AU)
endif()
foreach(kind Capture Probe)
 if(kind STREQUAL "Capture")
  set(code NfCp)
  set(name "NF Capture")
 else()
  set(code NfPr)
  set(name "NF Capture Probe")
 endif()
 juce_add_plugin(NF${kind} COMPANY_NAME "NF Audio Tools" BUNDLE_ID "audio.nf.${kind}" PLUGIN_MANUFACTURER_CODE NfAu PLUGIN_CODE ${code} FORMATS ${formats} PRODUCT_NAME "${name}" COPY_PLUGIN_AFTER_BUILD FALSE)
 if(kind STREQUAL "Capture")
  set_target_properties(NF${kind} PROPERTIES JUCE_PLUGIN_CODE NfCp JUCE_PRODUCT_NAME "NF Capture")
 else()
  set_target_properties(NF${kind} PROPERTIES JUCE_PLUGIN_CODE NfPr JUCE_PRODUCT_NAME "NF Capture Probe")
 endif()
 target_sources(NF${kind} PRIVATE Source/Plugin.cpp)
 target_compile_definitions(NF${kind} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 NF_PROBE=$<STREQUAL:${kind},Probe>)
 target_link_libraries(NF${kind} PRIVATE NFAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
endforeach()
juce_add_console_app(NFVerify PRODUCT_NAME "NF Verify")
target_sources(NFVerify PRIVATE Source/Verify.cpp)
target_compile_definitions(NFVerify PRIVATE NF_PROBE=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_include_directories(NFVerify PRIVATE JUCE/modules)
target_compile_definitions(NFVerify PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(NFVerify PRIVATE NFAssets NFCapture)
juce_add_console_app(NFVerifyProbe PRODUCT_NAME "NF Verify Probe")
target_sources(NFVerifyProbe PRIVATE Source/VerifyProbe.cpp)
target_compile_definitions(NFVerifyProbe PRIVATE NF_PROBE=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_include_directories(NFVerifyProbe PRIVATE JUCE/modules)
target_compile_definitions(NFVerifyProbe PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(NFVerifyProbe PRIVATE NFAssets NFProbe)
